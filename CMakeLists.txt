cmake_minimum_required(VERSION 3.25)
# Embedded debug information avoids dependence on an external PDB server during compilation.
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(BryceCompat VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "BryceCompat requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(brycecompat_core src/analyzer.cpp src/platform.cpp)
target_include_directories(brycecompat_core PUBLIC include third_party)
target_compile_definitions(brycecompat_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(brycecompat_core PUBLIC bcrypt advapi32 user32 shell32 shcore)
if(MSVC)
  target_compile_options(brycecompat_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
endif()
add_executable(brycecompat apps/cli.cpp)
target_link_libraries(brycecompat PRIVATE brycecompat_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE brycecompat_core)
  add_test(NAME core COMMAND core_tests)
  add_test(NAME help COMMAND brycecompat --help)
  add_test(NAME source_only COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/tools/check-source-only.ps1)
  add_executable(process_fixture WIN32 tests/process_fixture.cpp)
  add_test(NAME cli_integration COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.ps1 -Cli $<TARGET_FILE:brycecompat> -Fixture $<TARGET_FILE:process_fixture>)
endif()
install(TARGETS brycecompat RUNTIME DESTINATION .)
install(DIRECTORY profiles DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses/nlohmann-json)
set(CPACK_GENERATOR ZIP)
include(CPack)
